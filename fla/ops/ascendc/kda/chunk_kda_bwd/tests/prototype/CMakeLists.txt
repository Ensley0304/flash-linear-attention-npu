cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(kda_c32_prototype LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set_source_files_properties(gm_roundtrip.cpp PROPERTIES LANGUAGE ASC)
add_library(kda_gm_roundtrip SHARED gm_roundtrip.cpp)
target_compile_options(kda_gm_roundtrip PRIVATE --npu-arch=dav-2201 --cce-auto-sync=off -O2)
target_link_directories(kda_gm_roundtrip PRIVATE $ENV{ASCEND_HOME_PATH}/lib64)
target_link_libraries(kda_gm_roundtrip PRIVATE ascendc_runtime ascendcl runtime)
