cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(fwdh256_probe LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set_source_files_properties(cube_probe.cpp mix_probe.cpp prepare_budget.cpp PROPERTIES LANGUAGE ASC)
add_library(fwdh256_probe SHARED cube_probe.cpp mix_probe.cpp prepare_budget.cpp)
target_compile_options(fwdh256_probe PRIVATE --npu-arch=dav-2201 --cce-auto-sync=off -O2)
target_link_directories(fwdh256_probe PRIVATE $ENV{ASCEND_HOME_PATH}/lib64)
target_link_libraries(fwdh256_probe PRIVATE ascendc_runtime ascendcl runtime)
option(FWDH_RIGHT_RESIDENT "Keep S2 right operand in L1" OFF)
if(FWDH_RIGHT_RESIDENT)
    target_compile_definitions(fwdh256_probe PRIVATE FWDH_RIGHT_RESIDENT=1)
endif()
option(FWDH_D_BLOCK_PIPELINE "Publish S2 output blocks individually for Gh1" OFF)
if(FWDH_D_BLOCK_PIPELINE)
    target_compile_definitions(fwdh256_probe PRIVATE FWDH_D_BLOCK_PIPELINE=1)
endif()
