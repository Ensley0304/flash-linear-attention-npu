cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(fwdh256_probe LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 17)
set_source_files_properties(cube_probe.cpp mix_probe.cpp prepare_budget.cpp PROPERTIES LANGUAGE ASC)
add_library(fwdh256_probe SHARED cube_probe.cpp mix_probe.cpp prepare_budget.cpp)
target_compile_options(fwdh256_probe PRIVATE --npu-arch=dav-2201 --cce-auto-sync=off -O2)

target_link_libraries(fwdh256_probe PRIVATE ascendc_runtime ascendcl runtime)
